# Vault + CLI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship encrypted local vault + `flowrun vault *` CLI so `run`/`improve`/`verify` resolve secrets without plaintext in YAML or git.

**Architecture:** Thin `vault.ts` (scrypt + AES-256-GCM file under `.flowrun/vault.enc`) + `vault-keychain.ts` (macOS `security` / Linux `secret-tool`). `resolveVault(root)` is the single entry used by CLI and runner paths. Plain `FLOWRUN_VAULT` YAML remains a dev/CI override. Process-only unlock (no session file).

**Tech Stack:** TypeScript, Node 18+ `crypto`, Node built-in `node:test` + `tsx`, existing `yaml` dep, no new runtime deps.

**Spec:** `docs/superpowers/specs/2026-07-23-vault-design.md`

```ts
// packages/flowrun/src/vault-crypto.ts
import crypto from "node:crypto";

export type VaultFileV1 = {
  version: 1;
  kdf: "scrypt";
  salt: string; // base64
  iv: string; // base64
  ciphertext: string; // base64 (includes GCM auth tag at end — see below)
  meta: {
    created_at: string;
    updated_at: string;
    slot_count: number;
  };
};

const SCRYPT_N = 16384;
const SCRYPT_R = 8;
const SCRYPT_P = 1;
const KEY_LEN = 32;
const IV_LEN = 12;
const TAG_LEN = 16;

function deriveKey(passphrase: string, salt: Buffer): Buffer {
  return crypto.scryptSync(passphrase, salt, KEY_LEN, {
    N: SCRYPT_N,
    r: SCRYPT_R,
    p: SCRYPT_P,
  });
}

/** ciphertext field = base64( ciphertext_bytes || authTag ) */
export function sealVault(
  slots: Record<string, string>,
  passphrase: string,
  now = new Date(),
): VaultFileV1 {
  const salt = crypto.randomBytes(16);
  const iv = crypto.randomBytes(IV_LEN);
  const key = deriveKey(passphrase, salt);
  const plain = Buffer.from(JSON.stringify(slots), "utf8");
  const cipher = crypto.createCipheriv("aes-256-gcm", key, iv);
  const enc = Buffer.concat([cipher.update(plain), cipher.final()]);
  const tag = cipher.getAuthTag();
  const iso = now.toISOString();
  return {
    version: 1,
    kdf: "scrypt",
    salt: salt.toString("base64"),
    iv: iv.toString("base64"),
    ciphertext: Buffer.concat([enc, tag]).toString("base64"),
    meta: {
      created_at: iso,
      updated_at: iso,
      slot_count: Object.keys(slots).length,
    },
  };
}

export function openVault(
  file: VaultFileV1,
  passphrase: string,
): Record<string, string> {
  if (file.version !== 1 || file.kdf !== "scrypt") {
    throw new Error(`Unsupported vault format version=${file.version}`);
  }
  const salt = Buffer.from(file.salt, "base64");
  const iv = Buffer.from(file.iv, "base64");
  const blob = Buffer.from(file.ciphertext, "base64");
  if (blob.length <= TAG_LEN) throw new Error("Invalid vault ciphertext");
  const data = blob.subarray(0, blob.length - TAG_LEN);
  const tag = blob.subarray(blob.length - TAG_LEN);
  const key = deriveKey(passphrase, salt);
  try {
    const decipher = crypto.createDecipheriv("aes-256-gcm", key, iv);
    decipher.setAuthTag(tag);
    const plain = Buffer.concat([decipher.update(data), decipher.final()]);
    const obj = JSON.parse(plain.toString("utf8")) as Record<string, string>;
    if (!obj || typeof obj !== "object" || Array.isArray(obj)) {
      throw new Error("Invalid vault payload");
    }
    return obj;
  } catch {
    throw new Error("Vault decrypt failed (wrong passphrase or corrupted file)");
  }
}

/** Re-seal keeping original created_at */
export function resealVault(
  slots: Record<string, string>,
  passphrase: string,
  previous: VaultFileV1,
  now = new Date(),
): VaultFileV1 {
  const next = sealVault(slots, passphrase, now);
  next.meta.created_at = previous.meta.created_at;
  next.meta.updated_at = now.toISOString();
  next.meta.slot_count = Object.keys(slots).length;
  return next;
}
```

```ts
// packages/flowrun/src/vault.ts
import fs from "node:fs";
import path from "node:path";
import { parse as parseYaml } from "yaml";
import { defaultVaultPath, flowrunDir } from "./paths.js";
import {
  openVault,
  resealVault,
  sealVault,
  type VaultFileV1,
} from "./vault-crypto.js";
import {
  keychainDelete,
  keychainGet,
  keychainSet,
  keychainAvailable,
} from "./vault-keychain.js";

export type PassOpts = {
  root: string;
  passphrase?: string;
  vaultPath?: string;
};

export function resolveVaultPath(root: string): string {
  if (process.env.FLOWRUN_VAULT_PATH) {
    return path.resolve(process.env.FLOWRUN_VAULT_PATH);
  }
  return defaultVaultPath(root);
}

export function vaultExists(root: string): boolean {
  return fs.existsSync(resolveVaultPath(root));
}

function readVaultFile(vaultPath: string): VaultFileV1 {
  const raw = JSON.parse(fs.readFileSync(vaultPath, "utf8")) as VaultFileV1;
  return raw;
}

function writeVaultFile(vaultPath: string, file: VaultFileV1): void {
  fs.mkdirSync(path.dirname(vaultPath), { recursive: true });
  const tmp = `${vaultPath}.tmp`;
  fs.writeFileSync(tmp, JSON.stringify(file, null, 2), { mode: 0o600 });
  fs.renameSync(tmp, vaultPath);
  try {
    fs.chmodSync(vaultPath, 0o600);
  } catch {
    /* win */
  }
}

function requirePassphrase(opts: PassOpts): string {
  if (opts.passphrase) return opts.passphrase;
  if (process.env.FLOWRUN_VAULT_PASS) return process.env.FLOWRUN_VAULT_PASS;
  const vp = opts.vaultPath ?? resolveVaultPath(opts.root);
  const fromKc = keychainGet(vp);
  if (fromKc) return fromKc;
  throw new Error(
    "Vault passphrase required. Set FLOWRUN_VAULT_PASS, unlock keychain, or pass passphrase.",
  );
}

export function initVault(opts: PassOpts & { storeKeychain?: boolean }): string {
  const vaultPath = opts.vaultPath ?? resolveVaultPath(opts.root);
  if (fs.existsSync(vaultPath)) {
    throw new Error(`Vault already exists: ${vaultPath}`);
  }
  const pass = opts.passphrase ?? process.env.FLOWRUN_VAULT_PASS;
  if (!pass) throw new Error("Passphrase required for vault init");
  const file = sealVault({}, pass);
  writeVaultFile(vaultPath, file);
  if (opts.storeKeychain) {
    keychainSet(vaultPath, pass);
  }
  return vaultPath;
}

export function loadSlots(opts: PassOpts): Record<string, string> {
  const vaultPath = opts.vaultPath ?? resolveVaultPath(opts.root);
  if (!fs.existsSync(vaultPath)) {
    throw new Error(`No vault at ${vaultPath}. Run: flowrun vault init`);
  }
  const pass = requirePassphrase({ ...opts, vaultPath });
  return openVault(readVaultFile(vaultPath), pass);
}

function saveSlots(opts: PassOpts, slots: Record<string, string>): void {
  const vaultPath = opts.vaultPath ?? resolveVaultPath(opts.root);
  const pass = requirePassphrase({ ...opts, vaultPath });
  const prev = readVaultFile(vaultPath);
  // verify pass first
  openVault(prev, pass);
  const next = resealVault(slots, pass, prev);
  writeVaultFile(vaultPath, next);
}

export function setSlot(
  opts: PassOpts & { slot: string; value: string },
): void {
  if (!opts.slot?.trim()) throw new Error("slot name required");
  const slots = loadSlots(opts);
  slots[opts.slot] = opts.value;
  saveSlots(opts, slots);
}

export function getSlot(opts: PassOpts & { slot: string }): string {
  const slots = loadSlots(opts);
  if (!(opts.slot in slots)) throw new Error(`Missing vault slot: ${opts.slot}`);
  return slots[opts.slot];
}

export function listSlots(opts: PassOpts): string[] {
  return Object.keys(loadSlots(opts)).sort();
}

export function deleteSlot(opts: PassOpts & { slot: string }): void {
  const slots = loadSlots(opts);
  if (!(opts.slot in slots)) throw new Error(`Missing vault slot: ${opts.slot}`);
  delete slots[opts.slot];
  saveSlots(opts, slots);
}

export function vaultNeeds(
  required: string[],
  have: Record<string, string>,
): string[] {
  return required.filter((s) => !(s in have));
}

export type VaultStatus = {
  path: string;
  exists: boolean;
  source: "encrypted" | "yaml-override" | "none";
  slot_count?: number;
  keychain: boolean;
};

export function vaultStatus(root: string): VaultStatus {
  if (process.env.FLOWRUN_VAULT && fs.existsSync(process.env.FLOWRUN_VAULT)) {
    const data = parseYaml(fs.readFileSync(process.env.FLOWRUN_VAULT, "utf8")) as Record<
      string,
      string
    >;
    return {
      path: path.resolve(process.env.FLOWRUN_VAULT),
      exists: true,
      source: "yaml-override",
      slot_count: Object.keys(data ?? {}).length,
      keychain: keychainAvailable(),
    };
  }
  const p = resolveVaultPath(root);
  if (!fs.existsSync(p)) {
    return { path: p, exists: false, source: "none", keychain: keychainAvailable() };
  }
  const file = readVaultFile(p);
  return {
    path: p,
    exists: true,
    source: "encrypted",
    slot_count: file.meta?.slot_count,
    keychain: keychainAvailable(),
  };
}

/** Load secrets for run/improve/verify — never throw if no vault and nothing required */
export function resolveVault(opts: {
  root: string;
  passphrase?: string;
  prompt?: () => string;
}): Record<string, string> {
  if (process.env.FLOWRUN_VAULT && fs.existsSync(process.env.FLOWRUN_VAULT)) {
    const data = parseYaml(fs.readFileSync(process.env.FLOWRUN_VAULT, "utf8")) as Record<
      string,
      string
    >;
    return data ?? {};
  }

  const vaultPath = resolveVaultPath(opts.root);
  if (!fs.existsSync(vaultPath)) return {};

  let pass =
    opts.passphrase ??
    process.env.FLOWRUN_VAULT_PASS ??
    keychainGet(vaultPath) ??
    undefined;

  if (!pass && opts.prompt) {
    pass = opts.prompt();
  }
  if (!pass) {
    throw new Error(
      `Vault exists at ${vaultPath} but no passphrase. Set FLOWRUN_VAULT_PASS or run with a TTY to prompt.`,
    );
  }
  return openVault(readVaultFile(vaultPath), pass);
}

export function unlockVault(
  opts: PassOpts & { storeKeychain?: boolean },
): { ok: true; slot_count: number } {
  const slots = loadSlots(opts);
  const vaultPath = opts.vaultPath ?? resolveVaultPath(opts.root);
  if (opts.storeKeychain) {
    const pass = requirePassphrase({ ...opts, vaultPath });
    keychainSet(vaultPath, pass);
  }
  return { ok: true, slot_count: Object.keys(slots).length };
}

// re-export for tests that need dir
export { flowrunDir };
```

```ts
// packages/flowrun/src/vault-keychain.ts
import crypto from "node:crypto";
import { execFileSync } from "node:child_process";

const SERVICE = "flowrun";

export function keychainAccount(vaultPath: string): string {
  return crypto.createHash("sha256").update(vaultPath).digest("hex").slice(0, 16);
}

export function keychainAvailable(): boolean {
  if (process.platform === "darwin") return true;
  if (process.platform === "linux") {
    try {
      execFileSync("which", ["secret-tool"], { stdio: "ignore" });
      return true;
    } catch {
      return false;
    }
  }
  return false;
}

export function keychainGet(vaultPath: string): string | null {
  const account = keychainAccount(vaultPath);
  try {
    if (process.platform === "darwin") {
      const out = execFileSync(
        "security",
        ["find-generic-password", "-a", account, "-s", SERVICE, "-w"],
        { encoding: "utf8", stdio: ["ignore", "pipe", "ignore"] },
      );
      return out.trim() || null;
    }
    if (process.platform === "linux") {
      const out = execFileSync(
        "secret-tool",
        ["lookup", "service", SERVICE, "account", account],
        { encoding: "utf8", stdio: ["ignore", "pipe", "ignore"] },
      );
      return out.trim() || null;
    }
  } catch {
    return null;
  }
  return null;
}

export function keychainSet(vaultPath: string, passphrase: string): void {
  const account = keychainAccount(vaultPath);
  if (process.platform === "darwin") {
    try {
      execFileSync(
        "security",
        ["delete-generic-password", "-a", account, "-s", SERVICE],
        { stdio: "ignore" },
      );
    } catch {
      /* none */
    }
    execFileSync(
      "security",
      [
        "add-generic-password",
        "-a",
        account,
        "-s",
        SERVICE,
        "-w",
        passphrase,
        "-U",
      ],
      { stdio: "ignore" },
    );
    return;
  }
  if (process.platform === "linux") {
    execFileSync(
      "secret-tool",
      ["store", "--label", "flowrun vault", "service", SERVICE, "account", account],
      { input: passphrase, stdio: ["pipe", "ignore", "ignore"] },
    );
    return;
  }
  throw new Error("OS keychain not supported on this platform");
}

export function keychainDelete(vaultPath: string): void {
  const account = keychainAccount(vaultPath);
  try {
    if (process.platform === "darwin") {
      execFileSync(
        "security",
        ["delete-generic-password", "-a", account, "-s", SERVICE],
        { stdio: "ignore" },
      );
    } else if (process.platform === "linux") {
      execFileSync(
        "secret-tool",
        ["clear", "service", SERVICE, "account", account],
        { stdio: "ignore" },
      );
    }
  } catch {
    /* ignore */
  }
}
```

```ts
import * as vault from "./vault.js";
import readline from "node:readline";

async function promptSecret(label: string): Promise<string> {
  if (!process.stdin.isTTY) {
    throw new Error(`${label}: non-TTY; pass value as arg or use env`);
  }
  const rl = readline.createInterface({ input: process.stdin, output: process.stderr });
  // simple prompt (acceptable v0; optional: raw mode no-echo later)
  const answer = await new Promise<string>((resolve) => {
    rl.question(`${label}: `, (a) => {
      rl.close();
      resolve(a);
    });
  });
  return answer;
}
```

```ts
if (cmd === "vault") {
  const sub = rest[0];
  const vaultPath = vault.resolveVaultPath(root);
  const storeKeychain = Boolean(flags["store-keychain"]);
  const passStdin = Boolean(flags["passphrase-stdin"]);

  async function getPass(forInit = false): Promise<string> {
    if (process.env.FLOWRUN_VAULT_PASS) return process.env.FLOWRUN_VAULT_PASS;
    if (passStdin) {
      const chunks: Buffer[] = [];
      for await (const c of process.stdin) chunks.push(c as Buffer);
      return Buffer.concat(chunks).toString("utf8").trim();
    }
    if (process.stdin.isTTY) return promptSecret("Vault passphrase");
    throw new Error("Passphrase required (FLOWRUN_VAULT_PASS or --passphrase-stdin)");
  }

  if (sub === "init") {
    const passphrase = await getPass(true);
    const p = vault.initVault({ root, passphrase, storeKeychain });
    printJson({ ok: true, path: p });
    return;
  }
  if (sub === "unlock") {
    const passphrase = await getPass();
    const r = vault.unlockVault({ root, passphrase, storeKeychain });
    printJson({ ok: true, ...r, path: vaultPath });
    return;
  }
  if (sub === "set") {
    const slot = rest[1];
    let value = rest[2];
    if (!slot) {
      console.error("Usage: flowrun vault set <slot> [value]");
      process.exit(1);
    }
    if (value === undefined) value = await promptSecret(`Value for ${slot}`);
    const passphrase = await getPass();
    vault.setSlot({ root, passphrase, slot, value });
    printJson({ ok: true, slot }); // never print value
    return;
  }
  if (sub === "get") {
    const slot = rest[1];
    if (!slot) usage();
    const passphrase = await getPass();
    const value = vault.getSlot({ root, passphrase, slot });
    process.stdout.write(value + "\n"); // raw value only — humans
    return;
  }
  if (sub === "list") {
    const passphrase = await getPass();
    printJson(vault.listSlots({ root, passphrase }));
    return;
  }
  if (sub === "delete") {
    const slot = rest[1];
    if (!slot) usage();
    const passphrase = await getPass();
    vault.deleteSlot({ root, passphrase, slot });
    printJson({ ok: true, deleted: slot });
    return;
  }
  if (sub === "status") {
    printJson(vault.vaultStatus(root));
    return;
  }
  if (sub === "needs") {
    const ref = rest[1];
    let required: string[] = [];
    if (ref) {
      const rec = resolveRefFlexible(root, ref);
      if (!rec) {
        console.error(`Not found: ${ref}`);
        process.exit(2);
      }
      required = rec.workflow.requires_vault ?? [];
    } else {
      const all = loadAllWorkflows(root);
      required = [...new Set(all.flatMap((r) => r.workflow.requires_vault ?? []))];
    }
    let have: Record<string, string> = {};
    try {
      have = vault.resolveVault({ root });
    } catch {
      have = {};
    }
    const missing = vault.vaultNeeds(required, have);
    printJson({ required, missing, ok: missing.length === 0 });
    process.exit(missing.length ? 2 : 0);
  }
  console.error(`Unknown vault subcommand: ${sub}`);
  process.exit(1);
}
```
